max_framebuffers=2
disable_splash=1
boot_delay=0

# disable_fw_kms_setup=1: without it the firmware performs the display setup and the kernel inherits
# its choice, overscan included; with it mode setting is left to the kernel from the EDID (full KMS,
# dtoverlay=vc4-kms-v3d below). disable_overscan=1 turns off the firmware's overscan compensation;
# with KMS overscan is not applied by shrinking the framebuffer, so the framebuffer size proves nothing.
disable_fw_kms_setup=1 is the more consequential of the two: without it the
# FIRMWARE performs the display setup and the kernel inherits whatever it
# chose, overscan included. With it, mode setting is left to the kernel from
# the EDID, which is the intended arrangement when running full KMS
# (dtoverlay=vc4-kms-v3d, enabled below).
#
# disable_overscan=1 then turns off the firmware's overscan compensation
# itself. Note this is NOT contradicted by a framebuffer that measures a full
# 1920x1080: with KMS the overscan is not applied by shrinking the
# framebuffer, so the size alone proves nothing either way.
disable_fw_kms_setup=1
disable_overscan=1

# [pi3]: Raspberry Pi 3 is an accepted deploy target; without this section KMS is not enabled and the
# board falls back to the firmware framebuffer with software rendering. The vc4-kms-v3d overlay also
# drives VideoCore IV. A Pi 3 has no HEVC/H.265 hardware decode (the rpivid block exists only from
# BCM2711), no Vulkan and nothing above OpenGL ES 2.0; H.264 decode works through the bcm2835-codec
# V4L2 device.
[pi3]
dtoverlay=vc4-kms-v3d
dtparam=audio=on

[pi4]
dtoverlay=vc4-kms-v3d
dtparam=audio=on
# Pi 4 only: raises the SoC clock to 1.8GHz.
arm_boost=1

[pi5]
dtoverlay=vc4-kms-v3d-pi5
dtparam=audio=on
