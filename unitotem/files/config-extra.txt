max_framebuffers=2
disable_splash=1
boot_delay=0

# Both of these are in Raspberry Pi OS's own config.txt and were missing
# here, which showed up as a picture inset by black bands on a display that
# renders Raspberry Pi OS edge to edge on the same hardware.
#
# disable_fw_kms_setup=1 is the more consequential of the two: without it the
# FIRMWARE performs the display setup and the kernel inherits whatever it
# chose, overscan included. With it, mode setting is left to the kernel from
# the EDID, which is the intended arrangement when running full KMS
# (dtoverlay=vc4-kms-v3d, enabled below).
#
# disable_overscan=1 then turns off the firmware's overscan compensation
# itself. Note this is NOT contradicted by a framebuffer that measures a full
# 1920x1080: with KMS the overscan is not applied by shrinking the
# framebuffer, so the size alone proves nothing either way.
disable_fw_kms_setup=1
disable_overscan=1

# A [pi3] section is not optional padding: deploy/remote-deploy.sh accepts
# Raspberry Pi 3 as a target, and without this it matched no section at all,
# so KMS was never enabled there and the board fell back to the firmware
# framebuffer with software rendering - no acceleration of any kind. The same
# vc4-kms-v3d overlay drives VideoCore IV as well; the name is historical.
# Raspberry Pi OS in fact places this overlay in its general section for every
# model, so the per-model split here is the unusual choice, kept only because
# Pi 5 needs its own variant.
#
# What a Pi 3 still cannot do, regardless of configuration: HEVC/H.265
# hardware decode (the rpivid block only exists from BCM2711 onwards, so H.265
# falls back to software and 1080p is not realistically viewable), Vulkan, and
# anything above OpenGL ES 2.0. H.264 decode does work, through the same
# bcm2835-codec V4L2 device as on Pi 4.
[pi3]
dtoverlay=vc4-kms-v3d
dtparam=audio=on

[pi4]
dtoverlay=vc4-kms-v3d
dtparam=audio=on
# Pi 4 only - raises the SoC clock to 1.8GHz. Ignored on other models, but
# kept scoped since it means nothing to them. Present in Raspberry Pi OS.
arm_boost=1

[pi5]
dtoverlay=vc4-kms-v3d-pi5
dtparam=audio=on
