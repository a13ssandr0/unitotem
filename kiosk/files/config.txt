# NOTE: this file's real-world reach is narrower than it looks. Every device
# deployed through deploy/remote-deploy.sh (the only supported path - the
# old whole-disk dd flasher is gone) gets its actual boot-partition
# config.txt written at runtime by deploy/ota-engine.sh's
# cmd_set_default_recovery/cmd_set_trial_boot/cmd_promote_default, not by
# this file - it only ships as-is if someone dd's the built .img directly,
# bypassing remote-deploy.sh entirely (not a documented/supported path).
# kiosk/files/config-extra.txt is the actual live source of truth for the
# [pi4]/[pi5] dtoverlay/dtparam content below (ota-engine.sh appends it onto
# the boot-selection lines it owns) - keep the two in sync by hand, there is
# no templating between them.
[all]
arm_64bit=1
max_framebuffers=2
disable_splash=1
boot_delay=0
enable_uart=1
kernel=kernel8.img
initramfs initramfs8 followkernel

[pi4]
dtoverlay=vc4-kms-v3d
dtparam=audio=on

[pi5]
dtoverlay=vc4-kms-v3d-pi5
dtparam=audio=on

[all]
